{{/* escape-inline-code.html Escapes raw <, >, & inside inline tags. Block code (
) is protected and left untouched.
  Already-escaped entities (< > &) are preserved.
  Input: .Content (template.HTML)
  Output: template.HTML (safe, escaped)

  Strategy: protect 
 blocks -> split on  -> for each inline
  code span, apply "fix-after" escaping (escape all &, then restore
  known entities that were double-escaped) -> escape raw < and > ->
  restore 
 blocks.
*/}}
{{- $content := string . -}}

{{- /* Step 1: Protect block code regions from processing */ -}}
{{- $content = replace $content "" "BLOCK_PRE_END" -}}

{{- /* Step 2: Split on  to isolate each inline code span */ -}}
{{- $parts := split $content "" -}}
{{- $result := index $parts 0 -}}
{{- range $i, $part := after 1 $parts -}}
  {{- /* Each $part starts with the code content, followed by 
        and the rest. Split on  to separate them. */ -}}
  {{- $subparts := split $part "" -}}
  {{- $code := index $subparts 0 -}}

  {{- /* Step 3: Fix-after escaping.
        Order matters: escape & first, then un-break the entities we
        just double-escaped (e.g. < -> &lt; -> <),
        then escape raw < and >. */ -}}
  {{- $escaped := replace $code "&" "&" -}}
  {{- $escaped = replace $escaped "&lt;" "<" -}}
  {{- $escaped = replace $escaped "&gt;" ">" -}}
  {{- $escaped = replace $escaped "&amp;" "&" -}}
  {{- $escaped = replace $escaped "&quot;" """ -}}
  {{- $escaped = replace $escaped "&#39;" "'" -}}
  {{- $escaped = replace $escaped "<" "<" -}}
  {{- $escaped = replace $escaped ">" ">" -}}

  {{- /* Step 4: Reconstruct ... and append any trailing
        content after this code span. */ -}}
  {{- $result = printf "%s%s" $result $escaped -}}
  {{- if gt (len $subparts) 1 -}}
    {{- $result = printf "%s%s" $result (delimit (after 1 $subparts) "") -}}
  {{- end -}}
{{- end -}}

{{- /* Step 5: Restore protected block code regions */ -}}
{{- $result = replace $result "BLOCK_PRE_START" "" -}}

{{- $result | safeHTML -}}
